-- CREO TABELLA CPBI_UTENTI

-- creata tabella utenti
CREATE TABLE CPBI_UTENTI (
    UTENTE_ID           NUMBER NOT NULL,
    USERNAME            VARCHAR2(100 CHAR) NOT NULL,
    PASSWORD_HASH       VARCHAR2(255 CHAR) NOT NULL,
    RUOLO               VARCHAR2(20 CHAR) NOT NULL,
    NOME                VARCHAR2(100 CHAR),
    COGNOME             VARCHAR2(100 CHAR),
    EMAIL               VARCHAR2(255 CHAR),
    FLG_ATTIVO          NUMBER(1) DEFAULT 1 NOT NULL,
    DATA_CREAZIONE      TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
    DATA_ULTIMO_ACCESSO TIMESTAMP,

    CONSTRAINT PK_CPBI_UTENTI
        PRIMARY KEY (UTENTE_ID),

    CONSTRAINT UQ_CPBI_UTENTI_USERNAME
        UNIQUE (USERNAME),

    CONSTRAINT CK_CPBI_UTENTI_RUOLO
        CHECK (RUOLO IN ('SUPERUSER', 'VIEWER', 'ANALYST')),

    CONSTRAINT CK_CPBI_UTENTI_ATTIVO
        CHECK (FLG_ATTIVO IN (0, 1))
);

-- creata sequenza in oracle per gestire user_Id
CREATE SEQUENCE CPBI_UTENTI_SEQ
    START WITH 1
    INCREMENT BY 1
    NOCACHE;

-- creato trigger in oracle per aggiungere + 1 al numero della sequenza, in modo da avere sempre ultimo_numero +1 
CREATE OR REPLACE TRIGGER CPBI_UTENTI_BI
BEFORE INSERT ON CPBI_UTENTI
FOR EACH ROW
WHEN (NEW.UTENTE_ID IS NULL)
BEGIN
    :NEW.UTENTE_ID := CPBI_UTENTI_SEQ.NEXTVAL;
END;


-- tabella UTENTI|REPARTI
CREATE TABLE CPBI_UTENTI_REPARTI (
    UTENTE_ID   NUMBER NOT NULL,
    COD_REPARTO VARCHAR2(100 CHAR) NOT NULL,

    CONSTRAINT PK_CPBI_UTENTI_REPARTI
        PRIMARY KEY (UTENTE_ID, COD_REPARTO),

    CONSTRAINT FK_CPBI_UTENTI_REPARTI_UTENTE
        FOREIGN KEY (UTENTE_ID)
        REFERENCES CPBI_UTENTI (UTENTE_ID)
);
